"""Integration tests use only the one confirmed local frame, if present."""
import json
import sys
import unittest
from pathlib import Path
import numpy as np
ROOT=Path(__file__).resolve().parents[1]
sys.path[:0]=[str(ROOT),str(ROOT.parent/'20260903_fdp_baseline')]
from run_offline import load_case,load_reference,backend_result,verify_geometry,offline_audit
from bin_rim_pose.geometry import estimate
CASE=ROOT.parent/'20260903_fdp_baseline/cases/right_wrist_20260907_01_lingbot_operator1'
REFERENCE=ROOT.parent/'20260903_fdp_baseline/outputs/fdp_20260907_01_lingbot_operator1'


class ContractTests(unittest.TestCase):
    def test_rejected_candidate_is_not_published_as_pose(self):
        r={'ok':False,'candidate':{'T_base_link_from_4317_semantic':np.eye(4).tolist()},'rejection_reasons':['ambiguous_rim_planes']}
        official=backend_result(r,{},'camera')
        self.assertFalse(official['ok'])
        self.assertIsNone(official['matrices']['T_base_link_from_4317_semantic'])

    def test_network_guard(self):
        with self.assertRaisesRegex(RuntimeError,'network operation prohibited'):
            offline_audit('socket.connect',(None,('127.0.0.1',1234)))


@unittest.skipUnless((CASE/'manifest.json').exists() and REFERENCE.exists(),'frozen local assets unavailable')
class FrozenCaseTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.case=load_case(CASE);cls.cfg=json.loads((ROOT/'config/default.json').read_text())

    def test_archived_FDP_request_and_coordinates(self):
        _,checks=load_reference(REFERENCE,self.case)
        self.assertTrue(all(checks['archived_request_matches_case'].values()))
        self.assertTrue(checks['fdp_coordinate_replay_exact'])

    def test_different_mask_rejected_by_reference_binding(self):
        case=dict(self.case);case['mask']=case['mask'].copy();case['mask'][0,0]=~case['mask'][0,0]
        with self.assertRaisesRegex(ValueError,'archived input mismatch'):
            load_reference(REFERENCE,case)

    def test_real_frame_plane_ambiguity_retains_candidate_but_rejects(self):
        c=self.case
        r,_=estimate(c['rgb'],c['depth_m'],c['mask'],c['intrinsics'],c['base_from_camera'],self.cfg)
        self.assertFalse(r['ok'])
        self.assertIn('ambiguous_rim_planes',r['rejection_reasons'])
        self.assertIsNotNone(r['candidate'])
        self.assertTrue(any(a.get('unresolved_competitor') for a in r['plane_alternatives']))
        checks=verify_geometry(r['candidate'],c['intrinsics'],c['base_from_camera'],self.cfg)
        self.assertTrue(checks['ok'])
        self.assertIsNone(backend_result(r,checks,'camera')['matrices']['T_base_link_from_4317_semantic'])

if __name__=='__main__':unittest.main()
