#!/usr/bin/env bash


set -e


PROXY_HOST="192.168.211.229"
PROXY_PORT="7890"
PROXY_URL="http://${PROXY_HOST}:${PROXY_PORT}"


echo "[1/4] Checking mobile Clash proxy..."


if ! timeout 3 bash -c "</dev/tcp/${PROXY_HOST}/${PROXY_PORT}" 2>/dev/null; then
    echo "ERROR: Cannot connect to ${PROXY_HOST}:${PROXY_PORT}"
    echo "Check:"
    echo "  - USB tethering is connected"
    echo "  - Clash is running on the phone"
    echo "  - Allow LAN / proxy sharing is enabled"
    exit 1
fi


echo "[2/4] Stopping existing Codex Desktop processes..."


pkill -f '/usr/lib/chatgpt/resources/codex' 2>/dev/null || true
pkill -f '/usr/lib/chatgpt/ChatGPT' 2>/dev/null || true


sleep 1


echo "[3/4] Setting proxy environment..."


export HTTP_PROXY="${PROXY_URL}"
export HTTPS_PROXY="${PROXY_URL}"
export http_proxy="${PROXY_URL}"
export https_proxy="${PROXY_URL}"


# Localhost + laboratory LAN go directly, not through Clash
export NO_PROXY="localhost,127.0.0.1,::1,10.34.216.0/24,192.168.211.0/24"
export no_proxy="${NO_PROXY}"


echo "Proxy:    ${PROXY_URL}"
echo "No proxy: ${NO_PROXY}"


echo "[4/4] Starting Codex Desktop..."


exec /usr/bin/chatgpt \
    --proxy-server="${PROXY_URL}" \
    --proxy-bypass-list="localhost;127.0.0.1;::1;10.34.216.*;192.168.211.*;<local>"

## 12. 助手

一般不会有问题，这是 Linux 中很常见的配置：

```bash
export PATH="$HOME/bin:$PATH"
```

它的含义是：执行命令时，优先在你的 `~/bin` 中查找，然后再查找系统目录。

需要注意的主要是“同名命令覆盖”：如果 `~/bin` 中存在名为 `ssh`、`python`、`sudo` 等脚本，它会优先于系统命令执行。因此不要把来源不明的程序放进 `~/bin`。

可以检查目前里面有什么：

```bash
ls -la ~/bin
```

确认命令解析结果：

```bash
type -a codex-desktop-proxy
type -a ssh
```

当前配置不会影响 SSH 的网络路由，也不会自动启用代理；只有你运行 `codex-desktop-proxy` 时，代理设置才会传递给新启动的 Codex Desktop。

另外，它只影响之后新开的终端；当前终端需要运行：

```bash
source ~/.bashrc
```

或重新打开终端。
